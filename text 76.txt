-- ضع هذا الكود في LocalScript داخل StarterPlayerScripts أو StarterGui
-- السكريبت يقوم بتحديث القيم الأولى والثانية والثالثة والخامسة في جدول args
-- لتكون خاصة بنفس اللاعب (نفسه)، ثم يرسل الـ RemoteEvent

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()

-- الدالة الرئيسية: تحديث الأرج لتستهدف نفس اللاعب وإرسال الـ Remote
local function fireAtSelf()
    -- التأكد من وجود الشخصية
    local currentChar = player.Character
    if not currentChar then return end

    -- جلب الـ Humanoid الخاص باللاعب نفسه
    local selfHumanoid = currentChar:FindFirstChild("Humanoid")
    if not selfHumanoid then return end

    -- جلب الجزء المطلوب (LeftUpperArm) أو بديل
    local selfPart = currentChar:FindFirstChild("LeftUpperArm")
    if not selfPart then
        selfPart = currentChar:FindFirstChild("HumanoidRootPart") or currentChar:FindFirstChild("Torso")
    end
    if not selfPart then return end

    -- موقع الجزء الخاص بنفسك
    local selfPos = selfPart.Position

    -- الاتجاه: بما أننا نستهدف أنفسنا، سنستخدم اتجاه نظر الشخصية (LookVector) كقيمة افتراضية،
    -- أو يمكنك استخدام Vector3.new(0, 1, 0) إذا أردت.
    local myRoot = currentChar:FindFirstChild("HumanoidRootPart") or currentChar:FindFirstChild("Torso")
    local direction = Vector3.new(0, 1, 0) -- اتجاه احتياطي (لأعلى)
    if myRoot then
        direction = myRoot.CFrame.LookVector -- اتجاه أمام الشخصية
    end

    -- بناء جدول args الجديد مع تغيير الأرج الأول والثاني والثالث والخامس
    local newArgs = {
        selfPart,          -- الأول: الجزء الخاص بنفسك (LeftUpperArm)
        selfPos,           -- الثاني: موقع الجزء الخاص بنفسك
        direction,         -- الثالث: اتجاه الضربة (نفسك)
        "BatMelee",        -- الرابع: ثابت (اسم السلاح)
        selfHumanoid,      -- الخامس: الـ Humanoid الخاص بنفسك
        {                  -- السادس: إعدادات السلاح (نفسها بدون تغيير)
            EquipAnimation = true,
            AllowBlocking = true,
            Animations = {
                Hit2 = {
                    Id = "rbxassetid://117868238544491",
                    Priority = Enum.AnimationPriority.Action4
                },
                Idle = {
                    Id = "rbxassetid://87130013413201",
                    Priority = Enum.AnimationPriority.Action3
                },
                Hit = {
                    Id = "rbxassetid://100386746751382",
                    Priority = Enum.AnimationPriority.Action4
                },
                Equip = {
                    Id = "rbxassetid://83249421444656",
                    Priority = Enum.AnimationPriority.Action3
                },
                Block = {
                    Id = "rbxassetid://99523466744730",
                    Priority = Enum.AnimationPriority.Action4
                },
                Aim = {
                    Id = "rbxassetid://87130013413201",
                    Priority = Enum.AnimationPriority.Action3
                }
            },
            Ammo = 1/0,
            BurstCooldown = 0.2,
            MeleeHitSpeed = 1.3,
            Sounds = {
                MagOut = {
                    Id = "rbxassetid://90627076263265",
                    Volume = 0.5
                },
                MagIn = {
                    Id = "rbxassetid://122141314963769",
                    Volume = 0.5
                },
                Equip = {
                    Id = "rbxassetid://4549835866",
                    Volume = 0.5
                },
                Fire = {
                    Id = "rbxassetid://92874610447303",
                    Volume = 0.5,
                    RollOffMaxDistance = 500,
                    RollOffMinDistance = 10
                },
                BoltPull = {
                    Id = "rbxassetid://118952213304671",
                    Volume = 0.5
                },
                BoltRelease = {
                    Id = "rbxassetid://107899454072240",
                    Volume = 0.5
                }
            },
            IsLethal = false,
            ExtraAmmo = 1/0,
            Damage = 999999999999,
            GunViewmodels = false,
            IsKnife = false,
            BreakGlass = true,
            Rate = 1,
            HeadshotMultiplier = 1.5,
            AlwaysAiming = false,
            FiringModes = {
                "Semi"
            },
            TeamKilling = false,
            RandomizedDamage = true,
            Gore = true,
            EquipAnimationSpeed = 0.8,
            WeaponWeight = 0.94,
            LimbsMultiplier = 0.8
        }
    }

    -- إرسال الـ Remote بعد تعديل الأرج
    local remote = ReplicatedStorage:WaitForChild("GunSystemAssest"):WaitForChild("Packages"):WaitForChild("Knit"):WaitForChild("Services"):WaitForChild("CombatService"):WaitForChild("RE"):WaitForChild("Effect")
    remote:FireServer(unpack(newArgs))
end

-- مثال: استدعاء الدالة عند الضغط على زر E (يمكنك تغيير المفتاح)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.E then
        fireAtSelf()
    end
end)
